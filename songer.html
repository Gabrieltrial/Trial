function scanTok(t,o){

  const ops=[];let i=0,ts=0,chain=false,ws=-1;

  const flush=to=>{if(to>ts)ops.push({k:'t',s:t.slice(ts,to),from:ts,to})};

    if(boundary&&t.startsWith('sus',i)){

      const rest=t.slice(i+3);

      const run=(t.slice(i).match(/^[A-Za-zÀ-ÿ'’]+/)||[''])[0].toLowerCase().replace(/’/g,"'");

      if(!STOPSUS.some(p=>run.startsWith(p))){

        const letters=rest.replace(/[^A-Za-zÀ-ÿ].*$/,'');

        const wordAfter=letters.length>=2&&(/^[A-Z]/.test(letters)||VOWEL.test(letters));

        const chainAfter=rest[0]==='-'&&chainOK(rest.slice(1));

        if(wordAfter||chainAfter||(rest===''&&chain)){

          flush(i);ops.push({k:'s',from:i,to:i+3,free:(!wordAfter)});

          i+=3;ts=i;chain=true;ws=i;continue;

        }

      }

    }

    /* acord */

    if(/[A-G]/.test(c)){

      let o2=null;

      if(boundary) o2={glue2:o.glue2,first:o.first&&i===0,chain,mid:false};

      else if(/[a-zà-öø-ÿ]/.test(prev)&&!/^(?:Mc|Mac)$/.test(t.slice(0,i))) o2={glue2:o.glue2,first:false,chain:false,mid:true};

      if(o2){

        const ch=bChord(t.slice(i),o2);

        if(ch){

          flush(i);

          ops.push({k:'c',tok:ch,from:i,to:i+ch.length,def:(!/^[A-G][#b]?$/.test(ch))||o2.mid});

          i+=ch.length;ts=i;chain=true;continue;

        }

      }

    }

    /* guionets que només serveixen per enganxar acords */

    if(c==='-'){

      const nx=t.slice(i+1);

      if(ts===i&&chain){ts=i+1;i++;continue}

      if(/^[A-G]/.test(nx)&&bChord(nx,{glue2:o.glue2,first:false,chain:false,mid:false})){flush(i);ts=i+1;i++;continue}

      if(/^sus(?=$|-|[A-Z])/.test(nx)){flush(i);ts=i+1;i++;continue}

    }

    chain=false;i++;

  }

  flush(t.length);

  return ops;

}

}

      trail.forEach(t=>emb.push({pos:p,tok:t}));

    }

    maps.push({r:tk.r,len:tk.s.length,c0,cm});

  });

  clean=clean.replace(/\s+$/,'');

  if(!changed&&!emb.length) return null;

  const toClean=r=>{

    for(const m of maps){

      if(r<m.r) return m.c0;

      if(r<=m.r+m.len) return m.c0+m.cm[Math.min(r-m.r,m.len)];

    }

    return clean.length;

  };

  const Q=pc.map(c=>({pos:toClean(c.col),tok:c.tok}));

  const E=emb.map(e=>({pos:e.pos,tok:e.tok,used:false})),keep=[];

  Q.forEach(q=>{                                   // acords de la línia de sobre i de dins la lletra: sense duplicats

    const j=E.findIndex(e=>!e.used&&Math.abs(e.pos-q.pos)<=4&&e.tok[0]===q.tok[0]&&(e.tok.startsWith(q.tok)||q.tok.startsWith(e.tok)));

    if(j<0){keep.push(q);return}

    E[j].used=true;

    if(q.tok.length>E[j].tok.length)E[j].tok=q.tok;

  });

  const all=keep.concat(E).sort((a,b)=>a.pos-b.pos);

  let ch='';

  for(const c of all){const p=Math.max(c.pos,ch?ch.length+1:0);ch=ch.padEnd(p,' ')+c.tok}

  return {chord:ch,lyric:clean};

}


/* ---------- indicacions (intro, solo, 2x...): en cursiva, no són lletra ---------- */

const N_STRONG=/\b(?:intro|outro|solo|riffs?|interlude|instrumental|capo|fret|bar\s*chords?|chord\s*breaks?|tacet|arpeggios?|tremolo|ad[\s-]?lib|fade[\s-]?out|ritardando|repeat|second\s+guitar|lead\s+guitar|x\s?\d+|\d+\s?x)\b/i;

const N_START=/^(?:intro|outro|solo|riffs?|interlude|instrumental|capo|repeat|play|tacet|fade|bar\s*chords?|chord\s*break|second\s+guitar|lead\s+guitar|guitar\s+(?:solo|riff)|x\s?\d|\d+\s?x)\b/i;

const SECW=/^\s*[\[(]?\s*(?:verse|chorus|pre-?chorus|bridge|intro|outro|interlude|instrumental|solo|refrain|coda|estrofa|tornada|estribillo|introducci[oó])(?:\s*(?:\d+|x\d+|[ivx]{1,3}))*\s*[\])]?\s*:?\s*$/i;

function isSection(l){

  if(SECW.test(l)) return true;

  const m=l.match(/^\s*\[([^\]]{1,30})\]\s*$/);

  return !!m&&!CH.test(m[1].trim());

}

/* retorna el text de la indicació si la línia ho és, o null */

function noteText(l){

  const t=String(l||'').trim();

  if(!t) return null;

  if(/^#/.test(t)) return t.replace(/^#+\s*/,'');

  if(SECW.test(t)||isChordLine(t)) return null;

  const lab=t.match(/^[^\s:]{1,15}:\s+(\S.*)$/);

  if(lab&&isChordLine(lab[1])) return null;

  if(!N_STRONG.test(t)) return null;

  const inner=t.replace(/^[(\[]\s*/,'').replace(/\s*[)\]]$/,'').trim();

  const wrapped=/^[(\[]/.test(t)||/[)\]]$/.test(t);

  const caps=t===t.toUpperCase()&&/[A-Z]{3}/.test(t);

  return (wrapped||caps||N_START.test(inner))?inner:null;

}


/* ---------- neteja: elimina el que no és lletra ni acords + detecta títol i autor ---------- */

const LABEL_T=/^\s*(?:t[ií]tol|title|song|can[cç]ó|canci[oó]n|nom)\s*[:：]\s*(.+?)\s*$/i;

const LABEL_A=/^\s*(?:autor(?:a)?|artista?|author|by|int[èe]rpret|cantant|band|grup|group|composer|compositor|written\s+by|lletra\s+i\s+m[uú]sica)\s*[:：]\s*(.+?)\s*$/i;

const CHORDS_BY=/^(.{1,80}?)\s+(?:chords?(?:\s*&\s*lyrics)?|tabs?|acordes|acords|cifra|lyrics?)\s+(?:by|de|per)\s+(.{1,60})$/i;

const LABEL_J=/^\s*(?:key|tom|tonalitat|tonalidad|tono|capo|tuning|afinaci[oó]n?|tempo|bpm|difficulty|dificultat|dificultad|rating|votes?|views?|hits|album|[àa]lbum|year|any|a[ñn]o|genre|g[eè]nere|g[eé]nero|submitted\s+by|updated|last\s+update(?:d)?|last\s+edit(?:ed)?|transcribed\s+by|chords?\s+by|tabs?\s+by|strumming|chords?\s+used|acordes|acords|date|data|fecha|time\s+signature|comp[aà]s)\s*[:：]\s*.*$/i;

const J_WORD=/^\s*(?:chords?|tabs?|lyrics?|letra|acordes|acords|cifra|share|print|download|edit|favou?rites?|autoscroll|auto-?scroll|transpose|simplify|tuner|metronome|capo|report|reportar|compartir|imprimir|baixar|descargar|advertisement|ads?|sponsored|related|comments?|comentaris|rate|pro|premium|official|show\s+chords|hide\s+chords|chord\s+diagrams?|chord\s+chart|more|less|show\s+all|play|ver\s+v[ií]deo|aprender|tablature|ultimate[\s-]*guitar|cifra\s*club|chordie|songsterr|e-?chords)\s*$/i;

const J_MISC=[

 /^\s*[A-Ga-g]?[#b]?\s*\|[-\dhpbrxs\/\\~()^|*.\s]{4,}$/,

 /^\s*[xX\d]{6}\s*$/,

 /^\s*capo\s*(?:on\s*)?(?:the\s*)?\d+\w*\s*(?:fret)?\s*$/i,

 /^\s*key\s+of\s+\S+\s*$/i,

 /^\s*(?:bpm|tempo)\s*[=:]?\s*\d+\s*(?:bpm)?\s*$/i,

 /^\s*(?:standard|drop\s*[a-g]#?|half[\s-]*step\s+down)\s+tuning\s*$/i,

 /^\s*(?:submit|send|report)\s+(?:a\s+)?(?:correction|corrections|problem|error)\b.*$/i,

 /^\s*(?:chords?|acordes|acords)\s+(?:for|de|of)\s+.{1,80}$/i,

 /(?:https?:\/\/|www\.|@[\w-]+\.\w{2,}|\b[a-z0-9-]+\.(?:com|net|org|es|cat|br|io|fm)\b)/i,

 /(?:©|\(c\)\s*\d{4}|copyright|all\s+rights\s+reserved|todos\s+los\s+derechos|tots\s+els\s+drets)/i

];

function isJunk(l){

  if(!l||!l.trim()) return false;

  return LABEL_J.test(l)||J_WORD.test(l)||J_MISC.some(r=>r.test(l));

}

function isContentLine(l){

  return isChordLine(l)||isSection(l)||/\[[A-G][^\]]*\]/.test(l)||GARBLE.test(l)||!!parseMixed(l,false);

}

function cleanTitle(s){

  s=String(s||'').trim().replace(/^(?:chords?|tabs?|acordes?|acords?|cifra|lyrics?)\s+(?:for|de|of)\s+/i,'');

  let p;

  do{

    p=s;

    s=s.replace(/\s*[\(\[]?\s*\b(?:ver(?:si[oó]n)?\.?\s*\d+|official|chords?(?:\s*&\s*lyrics)?|tabs?|lyrics?|letra|cifra|acordes|acords)\s*[\)\]]?\s*$/i,'').trim();

  }while(s!==p);

  return s.replace(/^["“”'‘’]+|["“”'‘’]+$/g,'').replace(/[\s:,;-]+$/,'').trim();

}

function extractMeta(lines){

  const meta={},drop=new Set();

  const isDir=l=>/^\s*\{.*\}\s*$/.test(l);

  const cs=lines.findIndex(l=>l.trim()&&!isJunk(l)&&!isDir(l)&&noteText(l)===null&&isContentLine(l));

  const zoneEnd=cs>=0?cs:Math.min(lines.length,25);

  const idx=[];

  for(let i=0;i<zoneEnd;i++) if(lines[i].trim()&&!isDir(lines[i])) idx.push(i);

  const dropZone=cs>=0&&idx.length<=10;

  const cand=[];

  for(const i of idx){

    const l=lines[i].trim(); let m;

    if((m=l.match(LABEL_T))){if(!meta.title)meta.title=cleanTitle(m[1]);drop.add(i);continue}

    if((m=l.match(LABEL_A))){if(!meta.artist)meta.artist=cleanTitle(m[1]);drop.add(i);continue}

    if((m=l.match(CHORDS_BY))){

      if(!meta.title)meta.title=cleanTitle(m[1]);

      if(!meta.artist)meta.artist=cleanTitle(m[2]);

      drop.add(i);continue;

    }

    if(isJunk(lines[i])||noteText(lines[i])!==null){drop.add(i);continue}

    if(dropZone){

      if((m=l.match(/^[Bb]y\s+([A-ZÀ-Þ][^,.;!?]{1,40})$/))){if(!meta.artist)meta.artist=cleanTitle(m[1]);drop.add(i);continue}

      drop.add(i);

      if(l.length<=80) cand.push(l);

    }

  }

  if(dropZone&&cand.length){

    if(!meta.title){

      const c0=cand.shift();

      const m=c0.match(/^(.{1,70}?)\s+[-–—]\s+(.{1,50})$/);

      if(m){meta.title=cleanTitle(m[1]);if(!meta.artist)meta.artist=cleanTitle(m[2]);}

      else meta.title=cleanTitle(c0);

    }

    if(!meta.artist&&cand.length&&cand[0].length<=45) meta.artist=cleanTitle(cand[0]);

  }

  if(!meta.title)delete meta.title;

  if(!meta.artist)delete meta.artist;

  return {meta,drop};

}


function fixText(t){

  const meta={}; let n=0,removed=0,notes=0;

  t=t.replace(/\r\n?/g,'\n').replace(/\u00a0/g,' ').replace(/\t/g,'    ').replace(/\[\/?(?:ch|tab)\]/gi,'');

  let lines=t.split('\n').map(l=>l.replace(/\s+$/,''));

  // 1) títol i autor de la capçalera

  const ex=extractMeta(lines);

  Object.assign(meta,ex.meta);

  // 2) fora la capçalera i les línies que no són lletra ni acords

  const kept=[];

  lines.forEach((l,i)=>{

    if(ex.drop.has(i)||isJunk(l)){if(l.trim())removed++;return}

    kept.push(/^[\s\-_=*~.•·|]{3,}$/.test(l)?'':l);

  });

  lines=kept;

  // 3) acords a sobre de la lletra

  const plain=l=>!isChordLine(l)&&!SKIP.test(l)&&!/\[[^\]]+\]/.test(l)&&!GARBLE.test(l)&&noteText(l)===null;

  const lenient=lines.filter(l=>plain(l)&&parseMixed(l,false)).length>=2;

  const out=[]; let lastChord=false;

  for(const line of lines){

    const d=line.match(/^\s*\{\s*([a-z_]+)\s*:?\s*(.*?)\s*\}\s*$/i);   // directives ChordPro

    if(d){

      const k=d[1].toLowerCase(),v=d[2];

      if(k==='title'||k==='t') meta.title=v;

      else if(k==='artist'||k==='subtitle'||k==='st') meta.artist=v;

      else if(['comment','c','ci','comment_italic'].includes(k)){out.push('# '+v);notes++}

      lastChord=false;

      continue;

    }

    if(isChordLine(line)||SKIP.test(line)){out.push(line);lastChord=isChordLine(line);continue}

    const nt=noteText(line);

    if(nt!==null){if(nt!==''){out.push('# '+nt);notes++}lastChord=false;continue}

    if(/\[[^\]]+\]/.test(line)){

      const r=inlineToAbove(line);

      if(r){out.push(...r);n++;lastChord=false;continue}

    }

    const lab=line.match(/^\s*([^\s:]{1,15}:)\s+(\S.*)$/);               // "Intro: C G Am F"

    if(lab&&isChordLine(lab[2])){out.push(lab[1],lab[2]);n++;lastChord=false;continue}

    if(GARBLE.test(line)){                                                // còpia malmesa

      const P=lastChord?out[out.length-1]:null;

      const r=fuseLine(line,P);

      if(r){

        if(P) out.pop();

        if(r.chord.trim()) out.push(r.chord);

        if(r.lyric.trim()) out.push(r.lyric);

        n++; lastChord=!r.lyric.trim()&&!!r.chord.trim(); continue;

      }

    }

    const r=parseMixed(line,lenient);

    if(r){out.push(...r);n++;lastChord=false;continue}

    out.push(line); lastChord=false;

  }

  const text=out.join('\n').replace(/\n{3,}/g,'\n\n').replace(/^\n+|\n+$/g,'');

  return {text,meta,n,removed,notes};

}

/* hi ha res a netejar o convertir en aquest text? */

function needsFix(t){

  return String(t||'').split('\n').some(l=>/\[[A-G][^\]]*\]/.test(l)||(!isChordLine(l)&&GARBLE.test(l))||(!/^\s*#/.test(l)&&noteText(l)!==null));

}

function segHTML(c,tx){

  return '<span class="sg"><span class="ch">'+(c?esc(c):'&nbsp;')+'</span><span class="tx">'+(tx?esc(tx):'&nbsp;')+'</span></span>';

}

function pairHTML(cl,ll,n,useFlat){

  const ch=[]; const re=/\S+/g; let m;

  while((m=re.exec(cl))) ch.push({i:m.index,t:chordTok(m[0],n,useFlat)});

  ll=ll.replace(/\s+$/,'');

  let h='';

  const first=Math.min(ch[0].i,ll.length);

  if(first>0) h+=segHTML('',ll.slice(0,first));

  for(let k=0;k<ch.length;k++){

    const a=Math.min(ch[k].i,ll.length);

    const b=k+1<ch.length?Math.min(ch[k+1].i,ll.length):ll.length;

    h+=segHTML(ch[k].t,ll.slice(a,b));

  }

  return '<div class="ln"><span class="lw">'+h+'</span></div>';

}

function chordOnlyHTML(cl,n,useFlat){

  const h=cl.trim().split(/\s+/).map(t=>'<span class="ch inl">'+esc(chordTok(t,n,useFlat))+'</span>').join('');

  return '<div class="ln co"><span class="lw">'+h+'</span></div>';

}

function sheetHTML(text,n,useFlat){

  const L=String(text||'').replace(/\r/g,'').replace(/\t/g,'    ').split('\n').map(x=>x.replace(/\s+$/,''));

  let out='',i=0,blank=true;

  while(i<L.length){

    const l=L[i];

    if(!l.trim()){ if(!blank) out+='<div class="sp"></div>'; blank=true; i++; continue; }

    blank=false;

    if(isChordLine(l)){

      const nx=L[i+1];

      if(nx!==undefined&&nx.trim()&&!isChordLine(nx)&&!isSection(nx)&&noteText(nx)===null){

        out+=pairHTML(l,nx,n,useFlat); i+=2; continue;

      }

      out+=chordOnlyHTML(l,n,useFlat); i++; continue;

    }

    if(isSection(l)){

      out+='<div class="sec">'+esc(l.trim().replace(/^[\[(]\s*/,'').replace(/\s*[\])]$/,'').replace(/:$/,''))+'</div>'; i++; continue;

    }

    const nt=noteText(l);

    if(nt!==null){

      if(nt!=='') out+='<div class="note">'+esc(nt)+'</div>';

      i++; continue;

    }

    out+='<div class="ln pl">'+esc(l)+'</div>'; i++;

  }

  return out;

}

function fitSong(){

  const sh=$('#sheet'); if(!sh)return;

  const cs=getComputedStyle(sh);

  sh.classList.remove('wrap');

  sh.style.columnCount=1;

  const W=sh.clientWidth-parseFloat(cs.paddingLeft)-parseFloat(cs.paddingRight);

  const gap=parseFloat(cs.columnGap)||32;

  let mx=parseFloat(cs.fontSize)*14;

  sh.querySelectorAll('.lw').forEach(e=>{mx=Math.max(mx,e.getBoundingClientRect().width)});

  if(mx>W) sh.classList.add('wrap');

  const maxCols=Math.max(1,Math.min(3,Math.floor((W+gap)/(mx+gap))));

  let n;

  if(prefs.cols>0){

    n=Math.min(prefs.cols,maxCols);

  }else{

    const avail=Math.max(260,window.innerHeight-(sh.getBoundingClientRect().top+window.scrollY)-18);

    n=maxCols;

    for(let k=1;k<=maxCols;k++){

      sh.style.columnCount=k;

      if(sh.scrollHeight<=avail){n=k;break}

    }

  }

  sh.style.columnCount=n;

}

function onPaste(e){

  const cd=e.clipboardData||window.clipboardData; if(!cd)return;

  const txt=cd.getData('text/plain'); if(!txt)return;

  const r=fixText(txt);

  if(r.text===txt&&!Object.keys(r.meta).length) return;          // res a netejar: enganxat normal

  e.preventDefault();

  const ta=e.target;

  prevTxt=ta.value;

  prevTA={t:$('#t').value,a:$('#a').value};

  ta.focus();

  let done=false;

  try{done=document.execCommand('insertText',false,r.text)}catch(err){}

  if(!done){

    const s=ta.selectionStart,en=ta.selectionEnd;

    ta.value=ta.value.slice(0,s)+r.text+ta.value.slice(en);

    ta.selectionStart=ta.selectionEnd=s+r.text.length;

  }

  if(r.meta.title&&!$('#t').value.trim()) $('#t').value=r.meta.title;

  if(r.meta.artist&&!$('#a').value.trim()) $('#a').value=r.meta.artist;

  updLinks();

  showNote(r);

}

function showNote(r){

  const parts=[];

  if(r.n) parts.push('✔ '+r.n+' línies amb els acords posats a sobre de la lletra.');

  if(r.notes) parts.push('✔ '+r.notes+' indicacions (intro, solo…) posades en cursiva.');

  if(r.removed) parts.push('✔ '+r.removed+' línies que no eren lletra ni acords eliminades.');

  if(r.meta.title||r.meta.artist){

    const T=$('#t').value.trim(),A=$('#a').value.trim();

    parts.push('✔ Títol: <b>'+esc(T||'—')+'</b> · Autor: <b>'+esc(A||'—')+'</b> <a onclick="swapTA()">Intercanvia</a>');

  }

  const note=$('#note');

  if(!parts.length){note.style.display='none';return}

  note.innerHTML=parts.join('<br>')+' <a onclick="undoFix()">Desfés tot</a>';

  note.style.display='block';

}

function swapTA(){

  const t=$('#t'),a=$('#a'),v=t.value;

  t.value=a.value;a.value=v;updLinks();

  const note=$('#note');

  if(note) note.innerHTML=note.innerHTML.replace(/Títol: <b>.*?<\/b> · Autor: <b>.*?<\/b>/,'Títol: <b>'+esc(t.value||'—')+'</b> · Autor: <b>'+esc(a.value||'—')+'</b>');

}

function undoFix(){

  if(prevTxt!==null) $('#x').value=prevTxt;

  if(prevTA){$('#t').value=prevTA.t;$('#a').value=prevTA.a;updLinks()}

  prevTxt=null;prevTA=null;

  $('#note').style.display='none';

}

function updLinks(){

  const el=$('#links'); if(!el)return;

  const q=encodeURIComponent(($('#t').value+' '+$('#a').value).trim());

  const g=(extra)=>`https://www.google.com/search?q=${q}+${encodeURIComponent(extra)}`;

  el.innerHTML=[

   ['Ultimate Guitar',g('chords site:ultimate-guitar.com')],

   ['Cifraclub',g('site:cifraclub.com')],

   ['Chordie',g('site:chordie.com')],

   ['Google (lletra i acords)',g('lletra acords')]

  ].map(([n,u])=>`<a class="btn alt" style="text-decoration:none" target="_blank" rel="noopener" href="${u}">${n}</a>`).join('');

}

function saveSong(id){

  let text=$('#x').value.replace(/\r/g,'');

  if(needsFix(text)){const f=fixText(text);if(f.text.trim())text=f.text}

  const o={title:$('#t').value.trim()||'Sense títol',artist:$('#a').value.trim(),text};

  if(id){Object.assign(db.songs.find(s=>s.id===id),o)}else{id=uid();db.songs.push({id,...o})}

  save();go({name:'song',id});

}

function delSong(id){

  if(!confirm('Segur que vols esborrar aquesta cançó?'))return;

  db.songs=db.songs.filter(s=>s.id!==id);

  db.lists.forEach(l=>l.songs=l.songs.filter(x=>x!==id));

  save();go({name:'songs'});

}

function exportDb(){

  const b=new Blob([JSON.stringify(db)],{type:'application/json'});

  const a=document.createElement('a');a.href=URL.createObjectURL(b);

  a.download='cansoner-'+new Date().toISOString().slice(0,10)+'.json';a.click();

}

function importDb(inp){

  const f=inp.files[0];if(!f)return;

  const r=new FileReader();

  r.onload=()=>{try{const d=JSON.parse(r.result);

    if(!d.songs||!d.lists)throw 0;

    if(confirm('Això substituirà el contingut actual. Continuar?')){db=d;save();go({name:'songs'})}

  }catch(e){alert('Fitxer no vàlid')}};

  r.readAsText(f);

}

  const b=new Blob([html],{type:'text/html'});

  const a=document.createElement('a');

  a.href=URL.createObjectURL(b);a.download='cansoner.html';

  document.body.appendChild(a);a.click();a.remove();

  setTimeout(()=>URL.revokeObjectURL(a.href),1000);

}
